'; const suit = suits.find((item) => item.code === card.suit); const rank = card.rank === 'T' ? '10' : card.rank; return `

`; } function renderSeat(player) { const seat = document.getElementById(`seat-${player.index}`); const reveal = player.index === 0 || (handOver && !player.folded); const cards = player.hole.map((card) => cardMarkup(card, !reveal)).join(''); seat.className = `seat seat-${seatPositions[player.index]}${player.folded ?
